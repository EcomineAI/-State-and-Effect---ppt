import { useState } from "react";

export default function BatchCounter() {
  const [count, setCount] = useState(0);

  // BAD: all 3 read the same old count
  function badAdd() {
    setCount(count + 1); // 0 + 1
    setCount(count + 1); // 0 + 1
    setCount(count + 1); // 0 + 1 → result: +1 only
  }

  // GOOD: updater function gets the latest value
  function goodAdd() {
    setCount((c) => c + 1); // 0 → 1
    setCount((c) => c + 1); // 1 → 2
    setCount((c) => c + 1); // 2 → 3 → result: +3
  }

  // Either way, React batches the 3 calls into 1 re-render
  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={badAdd}>Bad +3</button>
      <button onClick={goodAdd}>Good +3</button>
    </div>
  );
}
